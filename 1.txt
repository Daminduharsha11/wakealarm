package com.example.powerscheduler

import android.app.AlarmManager
import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.util.Log

class AlarmReceiver : BroadcastReceiver() {

    companion object {
        private const val TAG = "AlarmReceiver"
    }

    override fun onReceive(context: Context, intent: Intent) {
        val action = intent.action
        if (action == AlarmManager.ACTION_NEXT_ALARM_CLOCK_CHANGED ||
            action == Intent.ACTION_BOOT_COMPLETED ||
            action == "android.intent.action.QUICKBOOT_POWERON") {
            
            Log.d(TAG, "Received broadcast action: $action. Re-evaluating next alarm...")

            val alarmManager = context.getSystemService(Context.ALARM_SERVICE) as AlarmManager
            val nextAlarmClock = alarmManager.nextAlarmClock

            if (nextAlarmClock != null) {
                val triggerTimeEpochMs = nextAlarmClock.triggerTime
                val triggerTimeEpochSec = triggerTimeEpochMs / 1000
                // Subtract a small buffer (e.g., 5 mins = 300s) so it wakes up slightly before the alarm rings
                val wakeEpochSec = triggerTimeEpochSec - 300

                Log.d(TAG, "Next alarm at $triggerTimeEpochSec. Setting RTC wake alarm for epoch $wakeEpochSec")

                val success = RootUtils.setRtcWakeAlarm(context, wakeEpochSec)
                Log.d(TAG, "RTC write success: $success")
            } else {
                Log.d(TAG, "No active next alarm found.")
            }
        }
    }
}

package com.example.powerscheduler

import android.app.AlarmManager
import android.content.Context
import android.os.Bundle
import android.widget.Button
import android.widget.TextView
import android.widget.Toast
import androidx.appcompat.app.AlertDialog
import androidx.appcompat.app.AppCompatActivity
import com.google.android.material.timepicker.MaterialTimePicker
import com.google.android.material.timepicker.TimeFormat
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

class MainActivity : AppCompatActivity() {

    private var shutdownHour = 23
    private var shutdownMinute = 0

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        val tvRootStatus = findViewById<TextView>(R.id.tvRootStatus)
        val tvNextAlarm = findViewById<TextView>(R.id.tvNextAlarm)
        val tvRtcStatus = findViewById<TextView>(R.id.tvRtcStatus)
        val btnSetShutdownTime = findViewById<Button>(R.id.btnSetShutdownTime)
        val btnTestRtc = findViewById<Button>(R.id.btnTestRtc)
        val btnManualShutdown = findViewById<Button>(R.id.btnManualShutdown)

        // 1. Verify Root & RTC on Startup and show friendly dialog
        val (rootOk, rootMsg) = RootUtils.checkRootAndRtc()
        if (rootOk) {
            tvRootStatus.text = "Root Status: Granted ✓"
            showStatusDialog("Root Access Confirmed", rootMsg)
        } else {
            tvRootStatus.text = "Root Status: Failed ✗"
            showStatusDialog("Root Warning", rootMsg)
        }

        // 2. Load system alarm status
        updateNextAlarmDisplay(tvNextAlarm, tvRtcStatus)

        // 3. Time Picker for Shutdown
        btnSetShutdownTime.setOnClickListener {
            val picker = MaterialTimePicker.Builder()
                .setTimeFormat(TimeFormat.CLOCK_24H)
                .setHour(shutdownHour)
                .setMinute(shutdownMinute)
                .setTitleText("Select Auto-Shutdown Time")
                .build()

            picker.addOnPositiveButtonClickListener {
                shutdownHour = picker.hour
                shutdownMinute = picker.minute
                val timeFormatted = String.format(Locale.getDefault(), "%02d:%02d", shutdownHour, shutdownMinute)
                btnSetShutdownTime.text = "Shutdown Time: $timeFormatted"
                Toast.makeText(this, "Shutdown scheduled for $timeFormatted", Toast.LENGTH_SHORT).show()
            }

            picker.show(supportFragmentManager, "SHUTDOWN_PICKER")
        }

        // 4. Manual RTC Test
        btnTestRtc.setOnClickListener {
            val testEpochSec = (System.currentTimeMillis() / 1000) + 3600
            val (success, message) = RootUtils.setRtcWakeAlarm(this, testEpochSec)
            Toast.makeText(this, if (success) "RTC Test Passed" else "RTC Test Failed", Toast.LENGTH_SHORT).show()
            tvRtcStatus.text = "Last Test: $message"
        }

        // 5. Manual Shutdown
        btnManualShutdown.setOnClickListener {
            AlertDialog.Builder(this)
                .setTitle("Confirm Shutdown")
                .setMessage("Are you sure you want to shut down the device now?")
                .setPositiveButton("Shutdown") { _, _ -> RootUtils.shutdownDevice() }
                .setNegativeButton("Cancel", null)
                .show()
        }
    }

    private fun showStatusDialog(title: String, message: String) {
        AlertDialog.Builder(this)
            .setTitle(title)
            .setMessage(message)
            .setPositiveButton("OK", null)
            .show()
    }

    private fun updateNextAlarmDisplay(tvNextAlarm: TextView, tvRtcStatus: TextView) {
        val alarmManager = getSystemService(Context.ALARM_SERVICE) as AlarmManager
        val nextAlarm = alarmManager.nextAlarmClock

        if (nextAlarm != null) {
            val sdf = SimpleDateFormat("EEE, dd MMM HH:mm", Locale.getDefault())
            val dateStr = sdf.format(Date(nextAlarm.triggerTime))
            tvNextAlarm.text = "Next System Alarm:\n$dateStr"
            
            val rtcEpochSec = (nextAlarm.triggerTime / 1000) - 300
            tvRtcStatus.text = "RTC Auto-Wake armed for:\n${sdf.format(Date(rtcEpochSec * 1000))}"
        } else {
            tvNextAlarm.text = "Next System Alarm: None set"
            tvRtcStatus.text = "RTC Auto-Wake: Inactive"
        }
    }
}

package com.example.powerscheduler

import android.content.Context
import android.util.Log
import java.io.File
import java.io.DataOutputStream

object RootUtils {
    private const val TAG = "RootUtils"

    fun checkRoot(): Pair<Boolean, String> {
        return try {
            val process = Runtime.getRuntime().exec(arrayOf("su", "-c", "id"))
            val exitCode = process.waitFor()
            if (exitCode == 0) {
                true to "Root access verified successfully (su working)!"
            } else {
                false to "Root permission denied or failed (Exit code: $exitCode)"
            }
        } catch (e: Exception) {
            Log.e(TAG, "Root check exception", e)
            false to "Root exception: ${e.localizedMessage}"
        }
    }

    private fun getExecutableBinary(context: Context): File? {
        val binFile = File(context.filesDir, "rtc_setter")
        try {
            // Always copy/refresh the binary from assets to ensure the latest compiled version is used
            context.assets.open("rtc_setter").use { input ->
                binFile.outputStream().use { output ->
                    input.copyTo(output)
                }
            }
            // Ensure execute permissions on the private file
            val chmodProcess = Runtime.getRuntime().exec(arrayOf("chmod", "755", binFile.absolutePath))
            chmodProcess.waitFor()
            return binFile
        } catch (e: Exception) {
            Log.e(TAG, "Failed to extract binary from assets", e)
            return null
        }
    }

    fun setRtcWakeAlarm(context: Context, epochSeconds: Long): Pair<Boolean, String> {
        return try {
            val binFile = getExecutableBinary(context) 
                ?: return false to "Failed to extract rtc_setter binary from assets"
            
            val process = Runtime.getRuntime().exec("su")
            val outputStream = DataOutputStream(process.outputStream)
            
            // Execute the compiled binary with root privileges
            val cmd = "${binFile.absolutePath} $epochSeconds\n"
            outputStream.writeBytes(cmd)
            outputStream.writeBytes("exit\n")
            outputStream.flush()
            
            val reader = process.inputStream.bufferedReader()
            val output = reader.readText()
            val errReader = process.errorStream.bufferedReader()
            val errorOutput = errReader.readText()
            
            process.waitFor()
            val exitCode = process.exitValue()
            
            Log.d(TAG, "Binary exec exit: $exitCode, out: $output, err: $errorOutput")
            
            if (exitCode == 0 && output.contains("SUCCESS")) {
                true to "RTC alarm set successfully via bundled binary for epoch $epochSeconds"
            } else {
                false to "Execution failed (Exit: $exitCode). Out: ${output.trim()} Err: ${errorOutput.trim()}"
            }
        } catch (e: Exception) {
            Log.e(TAG, "Failed to execute root command", e)
            false to "Exception: ${e.localizedMessage}"
        }
    }

    fun shutdownDevice(): Pair<Boolean, String> {
        return try {
            val process = Runtime.getRuntime().exec(arrayOf("su", "-c", "reboot -p"))
            val exitCode = process.waitFor()
            if (exitCode == 0) {
                Pair(true, "Shutdown command executed successfully")
            } else {
                Pair(false, "Shutdown failed with exit code $exitCode")
            }
        } catch (e: Exception) {
            Pair(false, "Shutdown exception: ${e.localizedMessage}")
        }
    }
    fun checkRootAndRtc(): Pair<Boolean, String> {
        return try {
            val process = Runtime.getRuntime().exec(arrayOf("su", "-c", "ls -l /dev/rtc0"))
            val output = process.inputStream.bufferedReader().readText().trim()
            val exitCode = process.waitFor()
            
            if (exitCode == 0 && output.isNotEmpty()) {
                Pair(true, "Root granted & RTC hardware ready ($output)")
            } else {
                Pair(false, "Root check passed, but /dev/rtc0 not accessible.")
            }
        } catch (e: Exception) {
            Pair(false, "Root access denied or unavailable: ${e.localizedMessage}")
        }
    }
}
